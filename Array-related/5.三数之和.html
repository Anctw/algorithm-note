三数之和：找出数组中所有和为 0 的不重复三元组
sort(nums.begin(), nums.end()); // 1. 排序
for(int k = 0; k < n - 2; k++) { // 2. 固定 k
  if(k > 0 && nums[k] == nums[k-1]) continue; // 去重
  l = k + 1; r = n - 1; // 3. 初始化双指针
  while(l < r) {
    int sum = nums[k] + nums[l] + nums[r];
    if(sum == 0) { ans.push_back({k, l, r}); ... }
    else if(sum < 0) l++; else r--;
  }
}
